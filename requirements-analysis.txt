# Offline analysis and plotting tools:
#   src/tools/plot_history.py · src/tools/field_testing/plot_data.py
#
#   .venv/bin/pip install -r requirements-analysis.txt
#
# Separate from requirements.txt because this is the expensive half. matplotlib
# and pandas pull numpy, pillow, contourpy, kiwisolver and fonttools; where a
# platform has no prebuilt wheel they compile from source, which is slow and
# needs a toolchain. Provisioning a radio does not need any of it, so the two
# sets are installed independently — and neither belongs on a Pi, which runs
# the stack in containers.
#
# Independent of requirements.txt on purpose: these tools read InfluxDB over
# HTTP and never speak the Meshtastic protocol, so they run on a machine with no
# meshtastic package at all.
#
# Lower bounds rather than exact pins: this is read-only analysis of data that
# already exists, so tracking a platform's available wheel matters more than
# reproducing one exact version. Pin here if a result ever has to be reproduced
# byte-for-byte.

matplotlib>=3.6      # plot_history.py, field_testing/plot_data.py
pandas>=2.0          # same
requests>=2.31       # plot_history.py queries the InfluxDB HTTP API directly
